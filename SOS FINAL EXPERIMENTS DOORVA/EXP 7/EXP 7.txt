#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>

int main()
{
    int fd;
    char buffer[100];

    printf("===== LINUX FILE OPERATIONS =====\n\n");

    // 1. Create/Open a file
    fd = open("sample.txt", O_CREAT | O_RDWR | O_TRUNC, 0644);

    if (fd == -1)
    {
        perror("Error opening file");
        return 1;
    }

    printf("1. File created/opened successfully.\n");

    // 2. Write to file
    char message[] =
        "Hello! This is a Linux file operations experiment.\n";

    if (write(fd, message, sizeof(message) - 1) == -1)
    {
        perror("Error writing to file");
        close(fd);
        return 1;
    }

    printf("2. Data written successfully.\n");

    // 3. Move file pointer to beginning
    lseek(fd, 0, SEEK_SET);

    // 4. Read from file
    int bytes = read(fd, buffer, sizeof(buffer) - 1);

    if (bytes == -1)
    {
        perror("Error reading file");
        close(fd);
        return 1;
    }

    buffer[bytes] = '\0';

    printf("3. Data read from file:\n");
    printf("%s", buffer);

    // 5. Close file
    close(fd);

    printf("4. File closed successfully.\n");

    // 6. Display file information
    struct stat fileInfo;

    if (stat("sample.txt", &fileInfo) == -1)
    {
        perror("stat failed");
        return 1;
    }

    printf("\n===== FILE INFORMATION =====\n");

    printf("File size: %ld bytes\n",
           fileInfo.st_size);

    printf("File permissions: %o\n",
           fileInfo.st_mode & 0777);

    printf("Number of links: %ld\n",
           fileInfo.st_nlink);

    // 7. Rename file
    if (rename("sample.txt", "renamed_sample.txt") == 0)
    {
        printf("\n5. File renamed successfully.\n");
    }
    else
    {
        perror("Error renaming file");
    }

    // 8. Delete file
    if (unlink("renamed_sample.txt") == 0)
    {
        printf("6. File deleted successfully.\n");
    }
    else
    {
        perror("Error deleting file");
    }

    printf("\nFile operations completed successfully.\n");

    return 0;
}
