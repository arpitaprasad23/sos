#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <semaphore.h>

#define BUFFER_SIZE 5

// Shared buffer structure
struct shared_data
{
    int buffer[BUFFER_SIZE];
    int in;
    int out;

    sem_t empty;
    sem_t full;
    sem_t mutex;
};

int main()
{
    int shmid;
    struct shared_data *shared;

    // Create shared memory
    shmid = shmget(IPC_PRIVATE,
                   sizeof(struct shared_data),
                   IPC_CREAT | 0666);

    if (shmid == -1)
    {
        perror("shmget failed");
        return 1;
    }

    // Attach shared memory
    shared = (struct shared_data *)shmat(shmid, NULL, 0);

    if (shared == (void *)-1)
    {
        perror("shmat failed");
        return 1;
    }

    // Initialize variables
    shared->in = 0;
    shared->out = 0;

    // Initialize semaphores
    // empty = number of empty spaces
    sem_init(&shared->empty, 1, BUFFER_SIZE);

    // full = number of filled spaces
    sem_init(&shared->full, 1, 0);

    // mutex = controls access to buffer
    sem_init(&shared->mutex, 1, 1);

    pid_t producer, consumer;

    // Create producer process
    producer = fork();

    if (producer == -1)
    {
        perror("fork failed");
        return 1;
    }

    if (producer == 0)
    {
        // Producer process

        for (int i = 1; i <= 10; i++)
        {
            // Wait for an empty slot
            sem_wait(&shared->empty);

            // Enter critical section
            sem_wait(&shared->mutex);

            shared->buffer[shared->in] = i;

            printf("Producer produced: %d\n", i);
            fflush(stdout);

            shared->in =
                (shared->in + 1) % BUFFER_SIZE;

            // Leave critical section
            sem_post(&shared->mutex);

            // Increase full count
            sem_post(&shared->full);

            sleep(1);
        }

        exit(0);
    }

    // Create consumer process
    consumer = fork();

    if (consumer == -1)
    {
        perror("fork failed");
        return 1;
    }

    if (consumer == 0)
    {
        // Consumer process

        for (int i = 1; i <= 10; i++)
        {
            // Wait for an available item
            sem_wait(&shared->full);

            // Enter critical section
            sem_wait(&shared->mutex);

            int item = shared->buffer[shared->out];

            printf("Consumer consumed: %d\n", item);
            fflush(stdout);

            shared->out =
                (shared->out + 1) % BUFFER_SIZE;

            // Leave critical section
            sem_post(&shared->mutex);

            // Increase empty count
            sem_post(&shared->empty);

            sleep(2);
        }

        exit(0);
    }

    // Parent waits for both processes
    wait(NULL);
    wait(NULL);

    // Destroy semaphores
    sem_destroy(&shared->empty);
    sem_destroy(&shared->full);
    sem_destroy(&shared->mutex);

    // Detach shared memory
    shmdt(shared);

    // Remove shared memory
    shmctl(shmid, IPC_RMID, NULL);

    printf("\nProducer-Consumer execution completed.\n");

    return 0;
}
