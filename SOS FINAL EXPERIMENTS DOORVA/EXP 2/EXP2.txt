#include <stdio.h>

struct Process {
    int pid;
    int arrival;
    int burst;
    int remaining;
    int completion;
    int turnaround;
    int waiting;
};

/* ---------- FCFS ---------- */

void ffs(struct Process p[], int n) {
    struct Process temp[20];
    int time = 0;

    for (int i = 0; i < n; i++)
        temp[i] = p[i];

    // Sort according to Arrival Time
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (temp[j].arrival > temp[j + 1].arrival) {
                struct Process t = temp[j];
                temp[j] = temp[j + 1];
                temp[j + 1] = t;
            }
        }
    }

    for (int i = 0; i < n; i++) {

        if (time < temp[i].arrival)
            time = temp[i].arrival;

        time += temp[i].burst;

        temp[i].completion = time;
        temp[i].turnaround =
            temp[i].completion - temp[i].arrival;

        temp[i].waiting =
            temp[i].turnaround - temp[i].burst;
    }

    printf("\n========== FCFS ==========\n");

    printf("PID\tAT\tBT\tCT\tTAT\tWT\n");

    float avgWT = 0, avgTAT = 0;

    for (int i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               temp[i].pid,
               temp[i].arrival,
               temp[i].burst,
               temp[i].completion,
               temp[i].turnaround,
               temp[i].waiting);

        avgWT += temp[i].waiting;
        avgTAT += temp[i].turnaround;
    }

    printf("Average Waiting Time    = %.2f\n", avgWT / n);
    printf("Average Turnaround Time = %.2f\n",
           avgTAT / n);
}


/* ---------- SJF ---------- */

void sjf(struct Process p[], int n) {

    struct Process temp[20];
    int completed[20] = {0};
    int time = 0;
    int count = 0;

    for (int i = 0; i < n; i++)
        temp[i] = p[i];

    while (count < n) {

        int index = -1;
        int shortest = 9999;

        // Find shortest available process
        for (int i = 0; i < n; i++) {

            if (!completed[i] &&
                temp[i].arrival <= time &&
                temp[i].burst < shortest) {

                shortest = temp[i].burst;
                index = i;
            }
        }

        // If no process has arrived
        if (index == -1) {
            time++;
            continue;
        }

        time += temp[index].burst;

        temp[index].completion = time;

        temp[index].turnaround =
            temp[index].completion -
            temp[index].arrival;

        temp[index].waiting =
            temp[index].turnaround -
            temp[index].burst;

        completed[index] = 1;
        count++;
    }

    printf("\n========== SJF ==========\n");

    printf("PID\tAT\tBT\tCT\tTAT\tWT\n");

    float avgWT = 0, avgTAT = 0;

    for (int i = 0; i < n; i++) {

        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               temp[i].pid,
               temp[i].arrival,
               temp[i].burst,
               temp[i].completion,
               temp[i].turnaround,
               temp[i].waiting);

        avgWT += temp[i].waiting;
        avgTAT += temp[i].turnaround;
    }

    printf("Average Waiting Time    = %.2f\n",
           avgWT / n);

    printf("Average Turnaround Time = %.2f\n",
           avgTAT / n);
}


/* ---------- ROUND ROBIN ---------- */

void RoundRobin(struct Process p[], int n, int quantum) {

    struct Process temp[20];

    int queue[100];
    int front = 0;
    int rear = 0;

    int visited[20] = {0};

    for (int i = 0; i < n; i++) {
        temp[i] = p[i];
        temp[i].remaining = temp[i].burst;
    }

    int time = 0;
    int completed = 0;

    // Start with first process
    queue[rear++] = 0;
    visited[0] = 1;

    printf("\n========== ROUND ROBIN ==========\n");

    while (completed < n) {

        // If queue is empty, find next process
        if (front == rear) {

            for (int i = 0; i < n; i++) {

                if (!visited[i] &&
                    temp[i].arrival <= time) {

                    queue[rear++] = i;
                    visited[i] = 1;
                }
            }

            if (front == rear) {
                time++;
                continue;
            }
        }

        int index = queue[front++];

        // Execute for time quantum
        int executionTime;

        if (temp[index].remaining < quantum)
            executionTime = temp[index].remaining;
        else
            executionTime = quantum;

        time += executionTime;

        temp[index].remaining -= executionTime;

        // Add newly arrived processes
        for (int i = 0; i < n; i++) {

            if (!visited[i] &&
                temp[i].arrival <= time) {

                queue[rear++] = i;
                visited[i] = 1;
            }
        }

        // If process is not finished
        if (temp[index].remaining > 0) {

            queue[rear++] = index;

        } else {

            temp[index].completion = time;

            temp[index].turnaround =
                temp[index].completion -
                temp[index].arrival;

            temp[index].waiting =
                temp[index].turnaround -
                temp[index].burst;

            completed++;
        }
    }

    printf("PID\tAT\tBT\tCT\tTAT\tWT\n");

    float avgWT = 0;
    float avgTAT = 0;

    for (int i = 0; i < n; i++) {

        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               temp[i].pid,
               temp[i].arrival,
               temp[i].burst,
               temp[i].completion,
               temp[i].turnaround,
               temp[i].waiting);

        avgWT += temp[i].waiting;
        avgTAT += temp[i].turnaround;
    }

    printf("Average Waiting Time    = %.2f\n",
           avgWT / n);

    printf("Average Turnaround Time = %.2f\n",
           avgTAT / n);
}


/* ---------- MAIN ---------- */

int main() {

    int n;
    int quantum;

    struct Process p[20];

    printf("===== CPU SCHEDULING ALGORITHMS =====\n");

    printf("\nEnter number of processes: ");
    scanf("%d", &n);

    for (int i = 0; i < n; i++) {

        p[i].pid = i + 1;

        printf("\nEnter Arrival Time for P%d: ",
               i + 1);
        scanf("%d", &p[i].arrival);

        printf("Enter Burst Time for P%d: ",
               i + 1);
        scanf("%d", &p[i].burst);
    }

    printf("\nEnter Time Quantum for Round Robin: ");
    scanf("%d", &quantum);

    /* Run all algorithms */

    fcfs(p, n);

    sjf(p, n);

    roundRobin(p, n, quantum);

    return 0;
}
