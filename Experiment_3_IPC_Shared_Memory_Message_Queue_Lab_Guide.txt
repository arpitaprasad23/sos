EXPERIMENT 3
Demonstrate Interprocess Communication (IPC) using Shared Memory and Message Queue

============================================================
IMPORTANT FOR LAB
============================================================
This guide assumes you are using a FRESH Kali Linux PC.
You do NOT need anything from any previous experiment.

There are TWO parts:
A) Shared Memory
B) Message Queue

For each part:
1. Create the programs
2. Compile them
3. Run the sender FIRST
4. Run the receiver/reader SECOND


============================================================
PART A - IPC USING SHARED MEMORY
============================================================

STEP 1 - CREATE SHARED MEMORY SENDER

Command:
nano shm_sender.c

Paste:

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ipc.h>
#include <sys/shm.h>

#define SHM_SIZE 1024

int main()
{
    key_t key = 1234;
    int shmid;
    char *shared_memory;

    // Create shared memory
    shmid = shmget(key, SHM_SIZE, IPC_CREAT | 0666);

    if (shmid == -1)
    {
        perror("shmget failed");
        return 1;
    }

    // Attach shared memory
    shared_memory = (char *)shmat(shmid, NULL, 0);

    if (shared_memory == (char *)-1)
    {
        perror("shmat failed");
        return 1;
    }

    printf("Enter message: ");
    fgets(shared_memory, SHM_SIZE, stdin);

    printf("Message written to shared memory.\n");

    // Detach shared memory
    shmdt(shared_memory);

    return 0;
}

------------------------------------------------------------
STEP 2 - CREATE SHARED MEMORY READER
------------------------------------------------------------

Command:
nano shm_reader.c

Paste:

#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/shm.h>

#define SHM_SIZE 1024

int main()
{
    key_t key = 1234;
    int shmid;
    char *shared_memory;

    // Access existing shared memory
    shmid = shmget(key, SHM_SIZE, 0666);

    if (shmid == -1)
    {
        perror("shmget failed");
        return 1;
    }

    // Attach shared memory
    shared_memory = (char *)shmat(shmid, NULL, 0);

    if (shared_memory == (char *)-1)
    {
        perror("shmat failed");
        return 1;
    }

    printf("Message received from shared memory:\n");
    printf("%s", shared_memory);

    // Detach shared memory
    shmdt(shared_memory);

    // Remove shared memory
    shmctl(shmid, IPC_RMID, NULL);

    return 0;
}

============================================================
PART B - IPC USING MESSAGE QUEUE
============================================================

STEP 1 - CREATE MESSAGE QUEUE SENDER

Command:
nano msg_sender.c

Paste:

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ipc.h>
#include <sys/msg.h>

struct message
{
    long message_type;
    char message_text[200];
};

int main()
{
    key_t key = 5678;
    int msgid;

    struct message msg;

    // Create message queue
    msgid = msgget(key, IPC_CREAT | 0666);

    if (msgid == -1)
    {
        perror("msgget failed");
        return 1;
    }

    msg.message_type = 1;

    printf("Enter message: ");
    fgets(msg.message_text, sizeof(msg.message_text), stdin);

    // Send message
    if (msgsnd(msgid,
               &msg,
               strlen(msg.message_text) + 1,
               0) == -1)
    {
        perror("msgsnd failed");
        return 1;
    }

    printf("Message sent successfully.\n");

    return 0;
}

------------------------------------------------------------
STEP 2 - CREATE MESSAGE QUEUE RECEIVER
------------------------------------------------------------

Command:
nano msg_receiver.c

Paste:

#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/msg.h>

struct message
{
    long message_type;
    char message_text[200];
};

int main()
{
    key_t key = 5678;
    int msgid;

    struct message msg;

    // Access message queue
    msgid = msgget(key, 0666);

    if (msgid == -1)
    {
        perror("msgget failed");
        return 1;
    }

    // Receive message
    if (msgrcv(msgid,
               &msg,
               sizeof(msg.message_text),
               1,
               0) == -1)
    {
        perror("msgrcv failed");
        return 1;
    }

    printf("Message received from message queue:\n");
    printf("%s", msg.message_text);

    // Remove message queue
    msgctl(msgid, IPC_RMID, NULL);

    return 0;
}

============================================================
WHAT THE EXPERIMENT DEMONSTRATES
============================================================

SHARED MEMORY:
Two processes communicate through a common memory segment.

Important functions:
shmget()  - creates/accesses shared memory
shmat()   - attaches shared memory
shmdt()   - detaches shared memory
shmctl()  - controls/removes shared memory

Key used in this program:
1234


MESSAGE QUEUE:
Processes communicate by sending and receiving messages through a queue.

Important functions:
msgget()  - creates/accesses message queue
msgsnd()  - sends message
msgrcv()  - receives message
msgctl()  - controls/removes message queue

Key used in this program:
5678
